> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在现有声明式 agent 层上构建可配置的多后台定时 agent 框架（cron + 独立 provider + 工具权限模型），首个实例 notes-organizer 定时整理 vault 子目录碎片笔记，产出可追溯报告与通知。

**Architecture:** 新增 `server/bgagent/` 功能层（scheduler/runner/notify），单向依赖 `server/agent/`。agent 层 5 处小改（permission 模型、auto_approve、write_scope/report_dir、ModelConfig 字段、round/done 事件）。配置为 config.yaml 的 `background_agents:` 列表。报告由编排层写 frontmatter .md 文件；通知走可插拔勾子（应用内 DB + sync 协议字段）。

**Tech Stack:** Python 3.12、FastAPI、asyncio、croniter（新依赖）、pytest、openai SDK。

**Spec:** `docs/superpowers/specs/2026-10-11-bgagent-design.md` —— 实现前先完整读一遍。

```python
"""Pre-execution validation of tool calls (pure functions, no side effects)."""
from __future__ import annotations

import re
import time
from datetime import datetime
from pathlib import Path

from pydantic import ValidationError

from .tools import TOOL_ARGS, TOOL_REGISTRY


def _extract_paths(args: dict) -> list[tuple[str, str]]:
    """Extract all path-like args, including nested move_docs moves[].

    Returns (label, value) pairs; label is used in error messages.
    """
    out: list[tuple[str, str]] = []
    for key in ("path", "from_path", "to_path"):
        v = args.get(key)
        if isinstance(v, str):
            out.append((key, v))
    moves = args.get("moves")
    if isinstance(moves, list):
        for i, item in enumerate(moves):
            if isinstance(item, dict):
                for key in ("from_path", "to_path"):
                    v = item.get(key)
                    if isinstance(v, str):
                        out.append((f"moves[{i}].{key}", v))
    return out


def validate_tool_call(name: str, args: dict, vault_dir: Path,
                       write_scope: Path | None = None,
                       report_dir: Path | None = None) -> str | None:
    """Validate args against the tool's Pydantic model + boundary/scope checks.

    ``write_scope`` / ``report_dir`` are resolved absolute paths; they only
    constrain tools with ``permission == "ask"`` (writes). Reads stay
    unrestricted so the agent can consult reports and the wider vault.
    """
    model = TOOL_ARGS.get(name)
    if model is None:
        return f"Error: unknown tool '{name}'"
    try:
        model(**args)
    except ValidationError as e:
        parts = []
        for err in e.errors():
            loc = ".".join(str(x) for x in err["loc"]) if err["loc"] else ""
            parts.append(f"{loc}: {err['msg']}" if loc else err["msg"])
        return f"Error: {'; '.join(parts)}"

    tool = TOOL_REGISTRY.get(name)
    is_ask = tool is not None and tool.permission == "ask"
    vault_resolved = vault_dir.resolve()

    for label, p in _extract_paths(args):
        full = (vault_dir / p).resolve()
        if not full.is_relative_to(vault_resolved):
            return f"Error: path '{p}' resolves outside vault"
        if is_ask and write_scope is not None and not full.is_relative_to(write_scope):
            return f"Error: path '{p}' ({label}) is outside the allowed write scope"
        if is_ask and report_dir is not None and full.is_relative_to(report_dir):
            return f"Error: path '{p}' ({label}) is in the protected report directory"
    return None


def parse_since(s: str) -> float:
    """Parse a human-friendly time string to a Unix timestamp.

    Supports: "5m", "1h", "2d", ISO 8601 ("2026-10-02T14:00:00"), or epoch seconds.
    Returns time.time() - delta for relative strings.
    """
    s = s.strip().lower()
    m = re.match(r"^(\d+)\s*([mhd])$", s)
    if m:
        n = int(m.group(1))
        unit = m.group(2)
        delta = n * {"m": 60, "h": 3600, "d": 86400}[unit]
        return time.time() - delta
    try:
        return float(s)
    except ValueError:
        pass
    try:
        dt = datetime.fromisoformat(s)
        return dt.timestamp()
    except ValueError:
        return 0.0
```

```python
@dataclass
class BackgroundAgentConfig:
    """One background scheduled agent (config.yaml 'background_agents' item)."""

    name: str
    alias: str
    enabled: bool
    cron: str
    target_dir: Path          # resolved absolute (vault_dir / target_dir)
    prompt: Path              # resolved absolute system-prompt file
    report_dir: Path          # resolved absolute (default <target_dir>/.reports/<alias>)
    max_rounds: int
    timeout_s: int
    model: ModelConfig


def load_background_agents(config_path: str, vault_dir: Path) -> list[BackgroundAgentConfig]:
    """Load and validate the 'background_agents' section.

    Fail-fast: raises ValueError on duplicate alias, invalid cron, missing
    prompt file, target_dir/report_dir outside the vault, or overlapping
    target_dirs. Returns [] when the section is absent.
    """
    from croniter import croniter

    path = Path(config_path)
    with open(path, "r", encoding="utf-8") as f:
        data = yaml.safe_load(f) or {}

    raw_list = data.get("background_agents")
    if not isinstance(raw_list, list) or not raw_list:
        return []

    base = path.parent
    vault_resolved = vault_dir.resolve()
    agents: list[BackgroundAgentConfig] = []

    for i, raw in enumerate(raw_list):
        if not isinstance(raw, dict):
            raise ValueError(f"background_agents[{i}] must be a mapping")

        alias = str(raw.get("alias", "")).strip()
        if not alias:
            raise ValueError(f"background_agents[{i}].alias is required")

        cron = str(raw.get("cron", "")).strip()
        if not croniter.is_valid(cron):
            raise ValueError(f"background_agents[{i}] ({alias}): invalid cron expression '{cron}'")

        target_rel = str(raw.get("target_dir", "")).strip()
        if not target_rel:
            raise ValueError(f"background_agents[{i}] ({alias}): target_dir is required")
        target = (vault_dir / target_rel).resolve()
        if not target.is_relative_to(vault_resolved):
            raise ValueError(f"background_agents[{i}] ({alias}): target_dir '{target_rel}' resolves outside vault")

        prompt_rel = str(raw.get("prompt", "")).strip()
        if not prompt_rel:
            raise ValueError(f"background_agents[{i}] ({alias}): prompt is required")
        prompt = (base / prompt_rel).resolve()
        if not prompt.is_file():
            raise ValueError(f"background_agents[{i}] ({alias}): prompt file not found: {prompt_rel}")

        report_rel = str(raw.get("report_dir", "")).strip()
        if report_rel:
            report_dir = (vault_dir / report_rel).resolve()
            if not report_dir.is_relative_to(vault_resolved):
                raise ValueError(f"background_agents[{i}] ({alias}): report_dir resolves outside vault")
        else:
            report_dir = target / ".reports" / alias

        model_raw = raw.get("model")
        if not isinstance(model_raw, dict) or "model_id" not in model_raw:
            raise ValueError(f"background_agents[{i}] ({alias}): model.model_id is required")

        agents.append(BackgroundAgentConfig(
            name=str(raw.get("name", alias)),
            alias=alias,
            enabled=bool(raw.get("enabled", True)),
            cron=cron,
            target_dir=target,
            prompt=prompt,
            report_dir=report_dir,
            max_rounds=int(raw.get("max_rounds", 30)),
            timeout_s=int(raw.get("timeout_s", 1800)),
            model=ModelConfig(
                model_id=str(model_raw["model_id"]),
                provider=str(model_raw.get("provider", "openai")),
                base_url=model_raw.get("base_url"),
                api_key=model_raw.get("api_key", ""),
                context_window=int(model_raw.get("context_window", 16384)),
                max_tokens=int(model_raw["max_tokens"]) if model_raw.get("max_tokens") is not None else None,
                temperature=float(model_raw["temperature"]) if model_raw.get("temperature") is not None else None,
            ),
        ))

    aliases = [a.alias for a in agents]
    if len(set(aliases)) != len(aliases):
        dupes = sorted({x for x in aliases if aliases.count(x) > 1})
        raise ValueError(f"duplicate background agent alias: {', '.join(dupes)}")

    for i in range(len(agents)):
        for j in range(i + 1, len(agents)):
            a, b = agents[i].target_dir, agents[j].target_dir
            if a.is_relative_to(b) or b.is_relative_to(a):
                raise ValueError(
                    f"background agents '{agents[i].alias}' and '{agents[j].alias}' "
                    f"have overlapping target_dir")

    return agents
```

```python
"""Notification hooks for background agent runs.

A hook is invoked once per finished run with a :class:`RunReport`.
First batch: InAppHook (DB row for the Web UI bell). The sync-protocol
delivery lives in the API layer (/api/sync reads the same table).
"""
from __future__ import annotations

import time
from dataclasses import dataclass


@dataclass
class RunReport:
    """Outcome of one background agent run."""

    alias: str
    name: str
    status: str            # ok | error | round_limit | timeout
    started_at: float
    finished_at: float
    report_path: str       # vault-relative posix path
    ops_total: int
    files_changed: int
    summary: str           # agent's final text, truncated


class NotificationHook:
    """Pluggable notification sink."""

    def notify(self, report: RunReport) -> None:
        raise NotImplementedError


class InAppHook(NotificationHook):
    """Insert a row into the notifications table (Web UI bell).

    Holds the shared ``_db_lock`` per codebase convention: every write
    path on the shared connection must hold it (the watcher thread is a
    concurrent writer).
    """

    def __init__(self, db_conn) -> None:
        self._conn = db_conn

    def notify(self, report: RunReport) -> None:
        from ..vault.db.schema import _db_lock

        with _db_lock:
            self._conn.execute(
                "INSERT INTO notifications (ts, alias, name, status, report_path, summary) "
                "VALUES (?, ?, ?, ?, ?, ?)",
                (time.time(), report.alias, report.name, report.status,
                 report.report_path, report.summary),
            )
            self._conn.commit()
```

```python
        self.agent = agent
        for e in self._events:
            yield e
        if self._exc is not None:
            raise self._exc
```

```python
"""One background agent run: AgentSpec -> event stream -> report file -> notify.

The orchestration layer (this module) writes the report, not the agent:
deterministic path, run metadata in frontmatter, agent's final text embedded
in the summary section. Write attribution goes through ``register_write`` so
the snapshot watcher logs ``source=<alias>``.
"""
from __future__ import annotations

import asyncio
import logging
import time
from datetime import datetime
from pathlib import Path

import yaml

from ..agent import Agent, AgentSpec, LoopRuntime, Runtime
from ..agent.tools import TOOL_REGISTRY
from ..config import BackgroundAgentConfig
from ..write_registry import register_write
from .notify import NotificationHook, RunReport

logger = logging.getLogger("md-vault.bgagent")

WRITE_TOOLS = {"write_doc", "edit_doc", "move_docs", "delete_doc"}
RETENTION = 30          # keep the N newest reports per alias


class _RunStats:
    """Accumulates one run's accounting from the event stream."""

    def __init__(self) -> None:
        self.rounds = 0
        self.round_texts: list[str] = []      # assistant text per round
        self.ops: list[dict] = []            # {ts, name, args_brief, result}
        self._open: dict[str, dict] = {}     # tool_call id -> op row
        self.ops_by_tool: dict[str, int] = {}
        self.files_changed: set[str] = set()
        self.prompt_tokens = 0               # peak of token_usage.used
        self.completion_tokens = 0           # sum of per-round used

    def record_call(self, data: dict) -> None:
        name = data["name"]
        op = {"ts": time.time(), "name": name,
              "args_brief": _brief_args(name, data.get("args") or {}),
              "result": ""}
        self.ops.append(op)
        self._open[data["id"]] = op
        self.ops_by_tool[name] = self.ops_by_tool.get(name, 0) + 1
        if name in WRITE_TOOLS:
            self.files_changed.update(_changed_paths(data.get("args") or {}))

    def record_result(self, data: dict) -> None:
        op = self._open.pop(data.get("id"), None)
        if op is not None:
            op["result"] = str(data.get("result", ""))[:200]

    def record_usage(self, used: int) -> None:
        self.prompt_tokens = max(self.prompt_tokens, used)
        self.completion_tokens += used

    @property
    def summary(self) -> str:
        for text in reversed(self.round_texts):
            if text.strip():
                return text.strip()
        return ""


def _brief_args(name: str, args: dict) -> str:
    """One-line human-readable arg summary for the ops table."""
    if name == "move_docs":
        return "; ".join(
            f"{m.get('from_path')} → {m.get('to_path')}"
            for m in args.get("moves", []) if isinstance(m, dict))
    if name in ("write_doc", "edit_doc", "delete_doc"):
        return str(args.get("path", ""))
    if name == "search_docs":
        return str(args.get("query", ""))[:80]
    if name == "recent_changes":
        return str(args.get("since", ""))
    if name == "get_related_docs":
        return str(args.get("path", ""))
    return ""


def _changed_paths(args: dict) -> set[str]:
    """Vault-relative paths touched by a write op (dedup input for files_changed)."""
    out: set[str] = set()
    for key in ("path", "to_path"):
        v = args.get(key)
        if isinstance(v, str):
            out.add(v)
    moves = args.get("moves")
    if isinstance(moves, list):
        for m in moves:
            if isinstance(m, dict) and isinstance(m.get("to_path"), str):
                out.add(m["to_path"])
    return out


def _prune_reports(report_dir: Path, vault: Path, alias: str) -> None:
    """Keep only the RETENTION newest reports; delete older ones (attributed).

    Filenames are timestamps (``YYYY-MM-DD_HHMM.md``) → lexicographic order
    == chronological order.
    """
    if not report_dir.is_dir():
        return
    files = sorted(report_dir.glob("*.md"))
    for fp in files[:-RETENTION] if len(files) > RETENTION else []:
        register_write(fp.relative_to(vault).as_posix(), alias)
        fp.unlink()


def _report_name(started_at: float) -> str:
    return datetime.fromtimestamp(started_at).strftime("%Y-%m-%d_%H%M") + ".md"


def _render_report(cfg: BackgroundAgentConfig, vault: Path, started_at: float,
                   finished_at: float, status: str, stats: _RunStats,
                   error_msg: str = "") -> tuple[str, str]:
    """Render the report markdown. Returns (markdown_text, summary_for_notify)."""
    target_rel = cfg.target_dir.relative_to(vault).as_posix()
    summary = stats.summary or ("（轮次耗尽，无总结）" if status == "round_limit" else "")
    if status == "timeout":
        summary = f"（运行超时：{error_msg}）"
    if status == "error":
        summary = f"（运行出错：{error_msg}）"

    ops_rows = "\n".join(
        f"| {datetime.fromtimestamp(o['ts']).strftime('%H:%M:%S')} "
        f"| {o['name']} | {o['args_brief']} | {o['result'] or '—'} |"
        for o in stats.ops) or "| — | — | — | 无操作 |"

    changed = "\n".join(f"- {p}" for p in sorted(stats.files_changed)) or "- 无"

    fm = {
        "run_id": datetime.fromtimestamp(started_at).strftime("%Y%m%dT%H%M%S"),
        "started_at": datetime.fromtimestamp(started_at).isoformat(timespec="seconds"),
        "finished_at": datetime.fromtimestamp(finished_at).isoformat(timespec="seconds"),
        "duration_s": int(finished_at - started_at),
        "rounds": stats.rounds,
        "prompt_tokens": stats.prompt_tokens,
        "completion_tokens": stats.completion_tokens,
        "target_dir": target_rel,
        "status": status,
        "alias": cfg.alias,
        "ops_total": len(stats.ops),
        "ops_by_tool": stats.ops_by_tool,
        "files_changed": len(stats.files_changed),
    }
    body = (
        f"---\n{yaml.safe_dump(fm, allow_unicode=True, sort_keys=False)}---\n\n"
        f"# 整理报告 {datetime.fromtimestamp(started_at).strftime('%Y-%m-%d %H:%M')}\n\n"
        f"## 总结（agent 原文）\n{summary or '（无）'}\n\n"
        f"## 操作清单\n| 时间 | 工具 | 参数 | 结果摘要 |\n| --- | --- | --- | --- |\n"
        f"{ops_rows}\n\n## 变更文件\n{changed}\n"
    )
    return body, summary


async def _consume(agent: Agent, message: str, stats: _RunStats) -> str:
    """Consume the event stream; return the done reason ('final' | 'round_limit')."""
    async for event in agent.chat(message):
        if event.type == "round":
            stats.rounds += 1
            stats.round_texts.append("")
        elif event.type == "token":
            if stats.round_texts:
                stats.round_texts[-1] += event.data.get("text") or ""
        elif event.type == "tool_call":
            stats.record_call(event.data)
        elif event.type == "tool_result":
            stats.record_result(event.data)
        elif event.type == "token_usage":
            stats.record_usage(int(event.data.get("used", 0)))
        elif event.type == "done":
            return event.data.get("reason", "final")
    return "final"


async def run_once(cfg: BackgroundAgentConfig, vault_dir: Path, *,
                   hooks: list[NotificationHook] | None = None,
                   runtime_cls: type[Runtime] = LoopRuntime) -> RunReport:
    """Run one background agent to completion; write the report; notify hooks.

    ``runtime_cls`` is injectable for tests (FakeRuntime). The run is bounded
    by ``cfg.timeout_s``; on timeout/cancel an error report is still written.
    """
    vault = vault_dir.resolve()
    started_at = time.time()
    stats = _RunStats()
    status, error_msg = "ok", ""

    _prune_reports(cfg.report_dir, vault, cfg.alias)

    spec = AgentSpec(
        system_prompt=cfg.prompt.read_text(encoding="utf-8"),
        model=cfg.model,
        tools=list(TOOL_REGISTRY.values()),
        max_rounds=cfg.max_rounds,
        vault_dir=vault,
        auto_approve=True,
        write_scope=cfg.target_dir,
        report_dir=cfg.report_dir,
    )
    agent = Agent(spec, runtime=runtime_cls())
    target_rel = cfg.target_dir.relative_to(vault).as_posix()

    try:
        reason = await asyncio.wait_for(
            _consume(agent, f"开始整理 {target_rel}/", stats),
            timeout=cfg.timeout_s)
        if reason == "round_limit":
            status = "round_limit"
    except asyncio.TimeoutError:
        status, error_msg = "timeout", f"exceeded {cfg.timeout_s}s"
    except asyncio.CancelledError:
        status, error_msg = "error", "cancelled"
        _write_report(cfg, vault, started_at, status, stats, error_msg)
        raise
    except Exception as e:
        status, error_msg = "error", f"{type(e).__name__}: {e}"
        logger.exception("[bg:%s] run failed", cfg.alias)

    finished_at = time.time()
    _write_report(cfg, vault, started_at, status, stats, error_msg)

    report_path = (cfg.report_dir / _report_name(started_at)).relative_to(vault).as_posix()
    report = RunReport(
        alias=cfg.alias, name=cfg.name, status=status,
        started_at=started_at, finished_at=finished_at,
        report_path=report_path, ops_total=len(stats.ops),
        files_changed=len(stats.files_changed), summary=stats.summary[:200],
    )
    for hook in hooks or []:
        hook.notify(report)
    logger.info("[bg:%s] run finished: status=%s ops=%d files=%d",
                cfg.alias, status, len(stats.ops), len(stats.files_changed))
    return report


def _write_report(cfg: BackgroundAgentConfig, vault: Path, started_at: float,
                  status: str, stats: _RunStats, error_msg: str) -> None:
    """Write the report file (mkdir + register_write + write_bytes)."""
    finished_at = time.time()
    body, _ = _render_report(cfg, vault, started_at, finished_at, status,
                             stats, error_msg)
    fp = cfg.report_dir / _report_name(started_at)
    fp.parent.mkdir(parents=True, exist_ok=True)
    register_write(fp.relative_to(vault).as_posix(), cfg.alias)
    fp.write_bytes(body.encode("utf-8"))
```

```python
"""Cron scheduler: one asyncio task per enabled background agent.

Runs in the same event loop as uvicorn (LLM calls are async IO). On each
tick: non-blocking lock acquire — if the previous run is still going, log
and skip (no queueing), then re-anchor from "now" (no catch-up bursts).
"""
from __future__ import annotations

import asyncio
import logging
from datetime import datetime
from pathlib import Path
from typing import Callable

from croniter import croniter

from ..config import BackgroundAgentConfig
from .notify import NotificationHook
from .runner import run_once

logger = logging.getLogger("md-vault.bgagent")


def _next_fire(cron: str, now: datetime | None = None) -> datetime:
    """Next cron trigger after ``now`` (local time, naive)."""
    now = now or datetime.now()
    return croniter.croniter(cron, now).get_next(datetime)


async def _handle_tick(cfg: BackgroundAgentConfig, vault_dir: Path,
                       lock: asyncio.Lock, hooks: list[NotificationHook],
                       run_fn: Callable) -> None:
    """One tick: non-blocking lock acquire, run or skip."""
    if not lock.acquire(blocking=False):
        logger.warning("[bg:%s] previous run still active, skipping tick", cfg.alias)
        return
    try:
        await run_fn(cfg, vault_dir, hooks=hooks)
    except asyncio.CancelledError:
        lock.release()
        raise
    except Exception:
        logger.exception("[bg:%s] run_once crashed (report may be missing)", cfg.alias)
    else:
        lock.release()


async def _agent_loop(cfg: BackgroundAgentConfig, vault_dir: Path,
                      lock: asyncio.Lock, hooks: list[NotificationHook],
                      run_fn: Callable) -> None:
    alias = cfg.alias
    while True:
        fire = _next_fire(cfg.cron)
        delay = (fire - datetime.now()).total_seconds()
        logger.info("[bg:%s] next run at %s (%.0fs)", alias, fire, delay)
        await asyncio.sleep(max(delay, 0))
        await _handle_tick(cfg, vault_dir, lock, hooks, run_fn)
        # loop top re-anchors from now — skipped/finished ticks never catch up


def start_background_agents(agents: list[BackgroundAgentConfig], vault_dir: Path,
                            hooks: list[NotificationHook] | None = None,
                            run_fn: Callable | None = None) -> list[asyncio.Task]:
    """Spawn one cron task per enabled agent. Must be called from a running loop."""
    run_fn = run_fn or run_once
    hooks = hooks or []
    tasks: list[asyncio.Task] = []
    for cfg in agents:
        if not cfg.enabled:
            continue
        lock = asyncio.Lock()
        tasks.append(asyncio.create_task(
            _agent_loop(cfg, vault_dir.resolve(), lock, hooks, run_fn),
            name=f"bgagent-{cfg.alias}"))
    return tasks
```

> 实现注意：`_handle_tick` 的锁释放用 `try/except/else` 而非 `finally`——acquire 失败时直接 return（不持锁），CancelledError 分支手动释放后 re-raise，正常/异常完成走 `else` 释放。不得用 `finally: lock.release()`（跳过分支会误释放别人的锁）。

```js
  async getNotifications(unread = true) {
    const res = await fetch('/api/notifications?unread=' + (unread ? 1 : 0))
    return res.json()
  },

  /** POST /api/notifications/read — {ids: [...]} 或 {all: true} */
  async markNotificationsRead(body) {
    const res = await fetch('/api/notifications/read', {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(body),
    })
    return res.json()
  },
```
